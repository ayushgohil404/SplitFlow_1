   108→   108→  // Direct expense: selected friends + email participants

   111→   111→  const [emailSplitAmounts, setEmailSplitAmounts] = useState<Record<string, string>>({});
   112→   112→  const [userSplitValue, setUserSplitValue] = useState('');
   113→   113→  const [newEmail, setNewEmail] = useState('');
   114→   114→  const [newName, setNewName] = useState('');
   115→   115→
   116→   116→  // Receipt upload
   117→   117→  const [receiptPreview, setReceiptPreview] = useState<string | null>(null);

   119→   119→
   120→   120→  // AI categorize
   121→   121→  const [categorizeLoading, setCategorizeLoading] = useState(false);
   122→   122→

   124→   124→  const [submitting, setSubmitting] = useState(false);
   125→   125→
   126→   126→  // Fetch groups
   127→   127→  useEffect(() => {
   128→   128→    async function fetchData() {

   171→   171→          if (me) setPaidById(me.id);
   172→   172→        }

   174→   174→        // silent
   175→   175→      } finally {
   176→   176→        setMembersLoading(false);
   177→   177→      }

   181→   181→
   182→   182→  // AI parse expense
   183→   183→  const handleNlpSubmit = async () => {
   184→   184→    if (!nlpInput.trim()) return;
   185→   185→    setNlpLoading(true);
   186→   186→    // Clear pending values from previous parse
   187→   187→    setPendingFriendSplitValues({});
   188→   188→    try {
   189→   189→      const res = await fetch('/api/ai/parse-expense', {
   190→   190→        method: 'POST',
   191→   191→        headers: { 'Content-Type': 'application/json' },
   192→   192→        body: JSON.stringify({ text: nlpInput }),
   193→   193→      });
   194→   194→      const data = await res.json();
   195→   195→
   196→   196→      // Handle HTTP errors and AI errors uniformly
   197→   197→      if (!res.ok && res.status !== 200) {
   198→   198→        toast.error(data.error || 'Could not parse expense. Try being more specific.');
   199→   199→        return;
   200→   200→      }
   201→   201→      if (data.error) {
   202→   202→        console.log('[AI Parse] Error response:', data);
   203→   203→        if (data.code === 'NOT_CONFIGURED') {
   204→   204→          toast.error('AI is not configured. Please set a valid GROQ_API_KEY in Vercel environment variables.');
   205→   205→        } else if (data.code === 'AI_ERROR') {
   206→   206→          toast.error(data.error);
   207→   207→        } else {
   208→   208→          toast.error(data.error || 'Could not parse expense. Try a different phrasing.');
   209→   209→        }
   210→   210→        return;
   211→   211→      }
   212→   212→
   213→   213→      // Set basic fields
   214→   214→      if (data.description) { setDescription(data.description); setDescError(''); }
   215→   215→      if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   216→   216→      if (data.category) setCategory(data.category);
   217→   217→      if (data.date) setDate(data.date);
   218→   218→
   219→   219→      const sType = data.splitType as string;
   220→   220→      const isExactOrPct = sType === 'exact' || sType === 'percentage';
   221→   221→
   222→   222→      // Handle group selection from AI parse
   223→   223→      if (data.groupId && typeof data.groupId === 'string') {
   224→   224→        const groupName = data.groupId.toLowerCase();
   225→   225→        const matchedGroup = groups.find((g) => g.name.toLowerCase() === groupName);
   226→   226→        if (matchedGroup) {
   227→   227→          setMode('group');
   228→   228→          setGroupId(matchedGroup.id);
   229→   229→          // Don't process friend/email splits for group expenses
   230→   230→          if (isExactOrPct) {
   231→   231→            setSplitType(sType as 'exact' | 'percentage');
   232→   232→          } else if (sType === 'equal' || sType === 'share') {
   233→   233→            setSplitType(sType === 'share' ? 'share' : 'equal');
   234→   234→          } else if (sType === 'single') {
   235→   235→            setSplitType('equal');
   236→   236→          }
   237→   237→          toast.success(`Expense parsed for group "${matchedGroup.name}"! Review and submit.`);
   238→   238→          return;
   239→   239→        }
   240→   240→        // Group name not found — fall through to direct mode
   241→   241→      }
   242→   242→
   243→   243→      // Determine mode: switch to direct if there are email splits or named non-me splits
   244→   244→      const emailSplits = data.emailSplits as { email: string; name?: string; amount?: number; percentage?: number }[] | undefined;
   245→   245→      const namedSplits = data.splits as { name: string; amount?: number | null; percentage?: number | null; share?: number }[] | undefined;
   246→   246→      const hasEmails = emailSplits && emailSplits.length > 0;
   247→   247→      const hasNamedSplits = namedSplits && namedSplits.some((s) => s.name !== 'me');
   248→   248→
   249→   249→      // Handle "single" / "only me" / personal expenses — no participants
   250→   250→      if (sType === 'single' && !hasEmails && !hasNamedSplits) {
   251→   251→        setMode('direct');
   252→   252→        setSelectedFriends([]);
   253→   253→        setEmailParticipants([]);
   254→   254→        setEmailSplitAmounts({});
   255→   255→        setPendingFriendSplitValues({});
   256→   256→        setSplitType('equal');
   257→   257→        toast.success('Expense parsed! No split needed — just submit.');
   258→   258→        return;
   259→   259→      }
   260→   260→
   261→   261→      // Extract "me" split value for exact/percentage modes
   262→   262→      if (isExactOrPct) {
   263→   263→        const meSplit = namedSplits?.find((s) => s.name === 'me');
   264→   264→        if (meSplit && (meSplit.amount != null || meSplit.percentage != null)) {
   265→   265→          setUserSplitValue(String(meSplit.amount ?? meSplit.percentage));
   266→   266→        } else if (data.amount) {
   267→   267→          const totalAmt = parseFloat(String(data.amount));
   268→   268→          let othersTotal = 0;
   269→   269→          if (sType === 'exact') {
   270→   270→            for (const ns of (namedSplits || []).filter((s) => s.name !== 'me')) othersTotal += ns.amount ?? 0;
   271→   271→            for (const es of (emailSplits || [])) othersTotal += es.amount ?? 0;
   272→   272→          } else {
   273→   273→            for (const ns of (namedSplits || []).filter((s) => s.name !== 'me')) othersTotal += ns.percentage ?? 0;
   274→   274→            for (const es of (emailSplits || [])) othersTotal += es.percentage ?? 0;
   275→   275→          }
   276→   276→          const myValue = sType === 'exact'
   277→   277→            ? Math.round((totalAmt - othersTotal) * 100) / 100
   278→   278→            : Math.round((100 - othersTotal) * 100) / 100;
   279→   279→          setUserSplitValue(myValue > 0 ? String(myValue) : '');
   280→   280→        } else {
   281→   281→          setUserSplitValue('');
   282→   282→        }
   283→   283→      } else {
   284→   284→        setUserSplitValue('');
   285→   285→      }
   286→   286→
   287→   287→      // Switch to direct mode when we have participants to add
   288→   288→      if (hasEmails || hasNamedSplits) {
   289→   289→        setMode('direct');
   290→   290→      }
   291→   291→
   292→   292→      // Process email splits
   293→   293→      if (hasEmails) {
   294→   294→        const newEmailParts: EmailParticipant[] = emailSplits.map((es) => ({
   295→   295→          email: es.email,
   296→   296→          name: es.name || es.email.split('@')[0],
   297→   297→        }));
   298→   298→        setEmailParticipants(newEmailParts);
   299→   299→
   300→   300→        if (isExactOrPct) {
   301→   301→          const amtMap: Record<string, string> = {};
   302→   302→          for (const es of emailSplits) {
   303→   303→            if (es.amount != null) amtMap[es.email] = String(es.amount);
   304→   304→            else if (es.percentage != null) amtMap[es.email] = String(es.percentage);
   305→   305→          }
   306→   306→          setEmailSplitAmounts(amtMap);
   307→   307→        } else {
   308→   308→          setEmailSplitAmounts({});
   309→   309→        }
   310→   310→      } else {
   311→   311→        setEmailParticipants([]);
   312→   312→        setEmailSplitAmounts({});
   313→   313→      }
   314→   314→
   315→   315→      // Process named splits — match to friends or add as email participants
   316→   316→      const friendAmtMap: Record<string, string> = {};
   317→   317→      const nonMeSplits = (namedSplits || []).filter((s) => s.name !== 'me');
   318→   318→
   319→   319→      for (const ns of nonMeSplits) {
   320→   320→        const friendMatch = friends.find(
   321→   321→          (f) =>
   322→   322→            f.name?.toLowerCase().trim() === ns.name?.toLowerCase().trim() ||
   323→   323→            f.email?.toLowerCase().split('@')[0] === ns.name?.toLowerCase().trim() ||
   324→   324→            f.email?.toLowerCase() === ns.name?.toLowerCase()
   325→   325→        );
   326→   326→        if (friendMatch) {
   327→   327→          if (!selectedFriends.includes(friendMatch.id)) {
   328→   328→            setSelectedFriends((prev) => [...prev, friendMatch.id]);
   329→   329→          }
   330→   330→          if (isExactOrPct && (ns.amount != null || ns.percentage != null)) {
   331→   331→            friendAmtMap[friendMatch.id] = String(ns.amount ?? ns.percentage);
   332→   332→          }
   333→   333→        } else if (ns.name?.includes('@')) {
   334→   334→          // Named person looks like an email — add as email participant
   335→   335→          const emailAddr = ns.name.toLowerCase();
   336→   336→          if (!emailSplits?.some((es) => es.email.toLowerCase() === emailAddr)) {
   337→   337→            setEmailParticipants((prev) => [...prev, { email: emailAddr, name: ns.name.split('@')[0] }]);
   338→   338→            if (isExactOrPct && (ns.amount != null || ns.percentage != null)) {
   339→   339→              setEmailSplitAmounts((prev) => ({ ...prev, [emailAddr]: String(ns.amount ?? ns.percentage) }));
   340→   340→            }
   341→   341→          }
   342→   342→        }
   343→   343→        // If not a friend and not an email, we skip them — user can add manually
   344→   344→        // (non-registered users need an email to track)
   345→   345→      }
   346→   346→
   347→   347→      if (Object.keys(friendAmtMap).length > 0) {
   348→   348→        setPendingFriendSplitValues(friendAmtMap);
   349→   349→      } else {
   350→   350→        setPendingFriendSplitValues({});
   351→   351→      }
   352→   352→
   353→   353→      // Handle split type
   354→   354→      if (isExactOrPct) {
   355→   355→        setSplitType(sType as 'exact' | 'percentage');
   356→   356→      } else if (sType === 'share') {
   357→   357→        setSplitType('share');
   358→   358→      } else if (sType === 'equal') {
   359→   359→        setSplitType('equal');
   360→   360→      } else if (sType === 'single') {
   361→   361→        setSplitType('equal');
   362→   362→      }
   363→   363→
   364→   364→      toast.success('Expense parsed! Review the details and submit.');
   365→   365→    } catch (err: any) {
   366→   366→      console.error('[AI Parse] Client error:', err);
   367→   367→      toast.error('Network error. Please check your connection and try again.');
   368→   368→    } finally {
   369→   369→      setNlpLoading(false);
   370→   370→    }
   371→   371→  };
   372→   372→
   373→   373→  // AI categorize
   374→   374→  const handleCategorize = async () => {
   375→   375→    if (!description.trim()) {
   376→   376→      toast.error('Enter a description first');
   377→   377→      return;
   378→   378→    }
   379→   379→    setCategorizeLoading(true);
   380→   380→    try {
   381→   381→      const res = await fetch('/api/ai/categorize', {
   382→   382→        method: 'POST',
   383→   383→        headers: { 'Content-Type': 'application/json' },
   384→   384→        body: JSON.stringify({ description }),
   385→   385→      });
   386→   386→      if (res.ok) {
   387→   387→        const data = await res.json();
   388→   388→        if (data.error) {
   389→   389→          toast.error(data.error || 'Failed to categorize');
   390→   390→        } else if (data.category) {
   391→   391→          setCategory(data.category);
   392→   392→          toast.success(`Categorized as ${data.category}`);
   393→   393→        }
   394→   394→      } else {
   395→   395→        const data = await res.json();
   396→   396→        toast.error(data.error || 'Failed to categorize');
   397→   397→      }
   398→   398→    } catch {
   399→   399→      toast.error('Failed to categorize');
   400→   400→    } finally {
   401→   401→      setCategorizeLoading(false);
   402→   402→    }
   403→   403→  };
   404→   404→
   405→   405→  // Receipt upload
   406→   406→  const handleReceiptUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   407→   407→    const file = e.target.files?.[0];
   408→   408→    if (!file) return;
   409→   409→    if (file.size > 10 * 1024 * 1024) {
   410→   410→      toast.error('Receipt image must be under 10MB');
   411→   411→      return;
   412→   412→    }
   413→   413→    setReceiptPreview(URL.createObjectURL(file));
   414→   414→    setReceiptLoading(true);
   415→   415→    try {
   416→   416→      const formData = new FormData();
   417→   417→      formData.append('receipt', file);
   418→   418→      const res = await fetch('/api/ai/receipt', { method: 'POST', body: formData });
   419→   419→      if (res.ok) {
   420→   420→        const data = await res.json();
   421→   421→        if (data.description) { setDescription(data.description); setDescError(''); }
   422→   422→        if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   423→   423→        if (data.category) setCategory(data.category);
   424→   424→        if (data.date) setDate(data.date);
   425→   425→        toast.success('Receipt scanned successfully!');
   426→   426→      } else {
   427→   427→        toast.error('Could not read receipt. Try a clearer photo.');
   428→   428→      }
   429→   429→    } catch {
   430→   430→      toast.error('Failed to scan receipt');
   431→   431→    } finally {
   432→   432→      setReceiptLoading(false);
   433→   433→    }
   434→   434→  };
   435→   435→
   436→   436→  // Add email participant
   437→   437→  const addEmailParticipant = () => {
   438→   438→    if (!newEmail.trim()) return;
   439→   439→    const email = newEmail.trim().toLowerCase();
   440→   440→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
   441→   441→    if (!emailRegex.test(email)) {
   442→   442→      toast.error('Please enter a valid email address');
   443→   443→      return;
   444→   444→    }
   445→   445→    if (emailParticipants.some((p) => p.email === email)) {
   446→   446→      toast.error('This email is already added');
   447→   447→      return;
   448→   448→    }
   449→   449→    setEmailParticipants([...emailParticipants, { email, name: newName.trim() || email.split('@')[0] }]);
   450→   450→    setNewEmail('');
   451→   451→    setNewName('');
   452→   452→  };
   453→   453→
   454→   454→  // Remove email participant
   455→   455→  const removeEmail = (email: string) => {
   456→   456→    setEmailParticipants(emailParticipants.filter((p) => p.email !== email));
