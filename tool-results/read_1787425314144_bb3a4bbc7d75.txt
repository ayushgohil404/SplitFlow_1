   465→

   478→      valid = false;
   479→    } else {
   480→      setAmountError('');
   481→    }

   483→      toast.error('Please select a group');
   484→      valid = false;

   489→    }

   514→          body.splits = splits.map((s) => ({
   515→            userId: s.userId,
   516→            share: s.share || 1,
   517→          }));
   518→        } else if (splitType !== 'equal') {
   519→          body.splits = splits.map((s) => ({
   520→            userId: s.userId,
   521→            value: parseFloat(s.value) || 0,

   523→        }

   528→          body.nonUserSplits = emailParticipants.map((p) => ({ email: p.email, name: p.name, share: 1 }));
   529→        } else if (splitType === 'share') {
   530→          body.splitType = 'equal';
   531→          body.splits = splits.map((s) => ({
   532→            userId: s.userId,
   533→            share: s.share || 1,
   534→          }));
   535→          body.nonUserSplits = emailParticipants.map((p) => ({ email: p.email, name: p.name, share: 1 }));
   536→        } else if (splitType === 'exact') {
   537→          // Include user's own split
   538→          const userAmt = parseFloat(userSplitValue) || 0;

   541→            bodySplits.push({ userId: user?.id, amount: userAmt });
   542→          }

   544→            userId: s.userId,
   545→            amount: parseFloat(s.value) || 0,
   546→          })));
   547→          body.splits = bodySplits;
   548→          // Include email participant amounts
   549→          if (emailParticipants.length > 0) {
   550→            body.nonUserSplits = emailParticipants.map((p) => ({
   551→              email: p.email,
   552→              name: p.name,
   553→              amount: parseFloat(emailSplitAmounts[p.email]) || 0,
   554→            }));
   555→          }
   556→        } else if (splitType === 'percentage') {
   557→          // Include user's own split
   558→          const userPct = parseFloat(userSplitValue) || 0;
   559→          const bodySplits: any[] = [];
   560→          if (userPct > 0) {
   561→            bodySplits.push({ userId: user?.id, percentage: userPct });
   562→          }
   563→          bodySplits.push(...splits.map((s) => ({
   564→            userId: s.userId,
   565→            percentage: parseFloat(s.value) || 0,
   566→          })));
   567→          body.splits = bodySplits;
   568→          // Include email participant percentages
   569→          if (emailParticipants.length > 0) {
   570→            body.nonUserSplits = emailParticipants.map((p) => ({
   571→              email: p.email,
   572→              name: p.name,
   573→              percentage: parseFloat(emailSplitAmounts[p.email]) || 0,

   586→        if (mode === 'group' && groupId) {
   587→          navigateToGroup(groupId);
   588→        } else {
   589→          setView('history');
   590→        }
   591→      } else {
   592→        const data = await res.json();
   593→        toast.error(data.error || 'Failed to add expense. Please try again.');
   594→      }
   595→    } catch {
   596→      toast.error('Network error. Please check your connection.');
   597→    } finally {
   598→      setSubmitting(false);

   600→  };
   601→
   602→  const perPerson = amount && !isNaN(parseFloat(amount)) && members.length > 0
   603→    ? (parseFloat(amount) / members.length).toFixed(2)
   604→    : '0.00';
   605→
   606→  // Computed total for direct exact/percentage split check
   607→  const totalSplitAmount = (() => {
   608→    const uv = parseFloat(userSplitValue) || 0;
   609→    const friendsTotal = splits.reduce((s, sp) => s + (parseFloat(sp.value) || 0), 0);
   610→    const emailTotal = Object.values(emailSplitAmounts).reduce((s, v) => s + (parseFloat(v) || 0), 0);
   611→    return uv + friendsTotal + emailTotal;
   612→  })();
