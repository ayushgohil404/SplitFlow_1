   427→   427→      const res = await fetch('/api/ai/receipt', { method: 'POST', body: formData });
   428→   428→      if (res.ok) {
   429→   429→        const data = await res.json();
   430→   430→        if (data.description) { setDescription(data.description); setDescError(''); }
   431→   431→        if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   432→   432→        if (data.category) setCategory(data.category);
   433→   433→        if (data.date) setDate(data.date);
   434→   434→        toast.success('Receipt scanned successfully!');
   435→   435→      } else {
   436→   436→        toast.error('Could not read receipt. Try a clearer photo.');
   437→   437→      }
   438→   438→    } catch {
   439→   439→      toast.error('Failed to scan receipt');

   441→   441→      setReceiptLoading(false);

   444→   444→
   445→   445→  // Add email participant
   446→   446→  const addEmailParticipant = () => {
   447→   447→    if (!newEmail.trim()) return;
   448→   448→    const email = newEmail.trim().toLowerCase();
   449→   449→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
   450→   450→    if (!emailRegex.test(email)) {
   451→   451→      toast.error('Please enter a valid email address');
   452→   452→      return;
   453→   453→    }
