    76→export function AddExpenseView() {
    77→  const { selectedGroupId, navigateToGroup, setView, user } = useAppStore();
    78→
    79→  // Expense mode
    80→  const [mode, setMode] = useState<ExpenseMode>(selectedGroupId ? 'group' : 'direct');
    81→
    82→  // Data
    83→  const [groups, setGroups] = useState<Group[]>([]);
    84→  const [members, setMembers] = useState<Member[]>([]);
    85→  const [friends, setFriends] = useState<Friend[]>([]);
    86→  const [membersLoading, setMembersLoading] = useState(false);
    87→
    88→  // AI NLP input
    89→  const [nlpInput, setNlpInput] = useState('');
    90→  const [nlpLoading, setNlpLoading] = useState(false);
    91→
    92→  // Form fields
    93→  const [description, setDescription] = useState('');
    94→  const [amount, setAmount] = useState('');
    95→  const [amountError, setAmountError] = useState('');
    96→  const [descError, setDescError] = useState('');
    97→  const [category, setCategory] = useState('general');
    98→  const [date, setDate] = useState(new Date().toISOString().split('T')[0]);
    99→  const [groupId, setGroupId] = useState(selectedGroupId || '');
   100→  const [paidById, setPaidById] = useState(user?.id || '');
   101→  const [splitType, setSplitType] = useState<'equal' | 'share' | 'exact' | 'percentage'>('equal');
   102→  const [splits, setSplits] = useState<{ userId: string; value: string; share: number }[]>([]);
   103→  const [note, setNote] = useState('');
   104→
   105→  // Direct expense: selected friends + email participants
   106→  const [selectedFriends, setSelectedFriends] = useState<string[]>([]);
   107→  const [emailParticipants, setEmailParticipants] = useState<EmailParticipant[]>([]);
   108→  const [emailSplitAmounts, setEmailSplitAmounts] = useState<Record<string, string>>({});
   109→  const [newEmail, setNewEmail] = useState('');
   110→  const [newName, setNewName] = useState('');
   111→
   112→  // Receipt upload
   113→  const [receiptPreview, setReceiptPreview] = useState<string | null>(null);
   114→  const [receiptLoading, setReceiptLoading] = useState(false);
   115→
   116→  // AI categorize
   117→  const [categorizeLoading, setCategorizeLoading] = useState(false);
   118→
   119→  // Submitting
   120→  const [submitting, setSubmitting] = useState(false);
   121→
   122→  // Fetch groups
   123→  useEffect(() => {
   124→    async function fetchData() {
   125→      try {
   126→        const [groupsRes, friendsRes] = await Promise.all([
   127→          fetch('/api/groups'),
   128→          fetch('/api/friends'),
   129→        ]);
   130→        if (groupsRes.ok) {
   131→          const data = await groupsRes.json();
   132→          const list = Array.isArray(data) ? data : data.groups || [];
   133→          setGroups(list);
   134→          if (!selectedGroupId && list.length > 0) {
   135→            setGroupId(list[0].id);
   136→          }
   137→        }
   138→        if (friendsRes.ok) {
   139→          const fData = await friendsRes.json();
   140→          setFriends(fData.friends || []);
   141→        }
   142→      } catch {
   143→        // silent
   144→      }
   145→    }
   146→    fetchData();
   147→  }, [selectedGroupId]);
   148→
   149→  // Fetch group members when group changes
   150→  useEffect(() => {
   151→    if (mode !== 'group' || !groupId) return;
   152→    setMembersLoading(true);
   153→    async function fetchMembers() {
   154→      try {
   155→        const res = await fetch(`/api/groups/${groupId}`);
   156→        if (res.ok) {
   157→          const data = await res.json();
   158→          const g = data.group || data;
   159→          const memberList = ((g.members || []) as any[]).map((m: any) => ({
   160→            id: m.user?.id || m.userId || m.id,
   161→            name: m.user?.name || m.name || 'Unknown',
   162→            email: m.user?.email || m.email || '',
   163→          }));
   164→          setMembers(memberList);
   165→          setSplits(memberList.map((m: Member) => ({ userId: m.id, value: '', share: 1 })));
   166→          const me = memberList.find((m: Member) => m.id === user?.id);
   167→          if (me) setPaidById(me.id);
   168→        }
   169→      } catch {
   170→        // silent
   171→      } finally {
   172→        setMembersLoading(false);
   173→      }
   174→    }
   175→    fetchMembers();
   176→  }, [groupId, mode, user?.id]);
   177→
   178→  // AI parse expense
   179→  const handleNlpSubmit = async () => {
   180→    if (!nlpInput.trim()) return;
   181→    setNlpLoading(true);
   182→    try {
   183→      const res = await fetch('/api/ai/parse-expense', {
   184→        method: 'POST',
   185→        headers: { 'Content-Type': 'application/json' },
   186→        body: JSON.stringify({ text: nlpInput }),
   187→      });
   188→      if (res.ok) {
   189→        const data = await res.json();
   190→        if (data.error) {
   191→          toast.error(data.error || 'Could not parse expense');
   192→          return;
   193→        }
   194→
   195→        // Set basic fields
   196→        if (data.description) { setDescription(data.description); setDescError(''); }
   197→        if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   198→        if (data.category) setCategory(data.category);
   199→        if (data.date) setDate(data.date);
   200→
   201→        // Determine mode: switch to direct if there are email splits
   202→        const emailSplits = data.emailSplits as { email: string; name?: string; amount?: number; percentage?: number }[] | undefined;
   203→        const namedSplits = data.splits as { name: string; amount?: number | null; percentage?: number | null }[] | undefined;
   204→        const hasEmails = emailSplits && emailSplits.length > 0;
   205→        const hasNamedSplits = namedSplits && namedSplits.some((s) => s.name !== 'me');
   206→
   207→        if (hasEmails) {
   208→          // Switch to direct mode for email participants
   209→          setMode('direct');
   210→
   211→          // Add email participants and store their split amounts
   212→          const newEmailParts: EmailParticipant[] = emailSplits.map((es) => ({
   213→            email: es.email,
   214→            name: es.name || es.email.split('@')[0],
   215→          }));
   216→          setEmailParticipants(newEmailParts);
   217→
   218→          // Store email split amounts for exact/percentage
   219→          if (sType === 'exact' || sType === 'percentage') {
   220→            const amtMap: Record<string, string> = {};
   221→            for (const es of emailSplits) {
   222→              if (es.amount != null) amtMap[es.email] = String(es.amount);
   223→              else if (es.percentage != null) amtMap[es.email] = String(es.percentage);
   224→            }
   225→            setEmailSplitAmounts(amtMap);
   226→          } else {
   227→            setEmailSplitAmounts({});
   228→          }
   229→
   230→          // Try to match named splits (non-'me') to friends
   231→          if (hasNamedSplits) {
   232→            const nonMeSplits = namedSplits.filter((s) => s.name !== 'me');
   233→            for (const ns of nonMeSplits) {
   234→              const friendMatch = friends.find(
   235→                (f) => f.name?.toLowerCase() === ns.name?.toLowerCase() || f.email?.toLowerCase() === ns.name?.toLowerCase()
   236→              );
   237→              if (friendMatch && !selectedFriends.includes(friendMatch.id)) {
   238→                setSelectedFriends((prev) => [...prev, friendMatch.id]);
   239→              }
   240→            }
   241→          }
   242→        } else if (hasNamedSplits) {
   243→          // No emails, just named people — try to match to friends in direct mode
   244→          setMode('direct');
   245→          const nonMeSplits = namedSplits.filter((s) => s.name !== 'me');
   246→          for (const ns of nonMeSplits) {
   247→            const friendMatch = friends.find(
   248→              (f) => f.name?.toLowerCase() === ns.name?.toLowerCase() || f.email?.toLowerCase() === ns.name?.toLowerCase()
   249→            );
   250→            if (friendMatch && !selectedFriends.includes(friendMatch.id)) {
   251→              setSelectedFriends((prev) => [...prev, friendMatch.id]);
   252→            }
   253→          }
   254→          // For named people not found as friends, add as email participants if they look like emails
   255→          for (const ns of nonMeSplits) {
   256→            if (ns.name?.includes('@') && !emailParticipants.some((ep) => ep.email === ns.name.toLowerCase())) {
   257→              setEmailParticipants((prev) => [...prev, { email: ns.name.toLowerCase(), name: ns.name.split('@')[0] }]);
   258→            }
   259→          }
   260→        }
   261→
   262→        // Handle split type and populate split values
   263→        const sType = data.splitType as string;
   264→        if (sType === 'exact' || sType === 'percentage') {
   265→          setSplitType(sType as 'exact' | 'percentage');
   266→          // For exact/percentage with named friends, populate the split values
   267→          if (hasNamedSplits && !hasEmails) {
   268→            // Will be populated by useEffect when selectedFriends changes
   269→          }
   270→        } else if (sType === 'equal') {
   271→          setSplitType('equal');
   272→        } else if (sType === 'single') {
   273→          setSplitType('equal');
   274→          // Single = just the user, no participants needed
   275→        }
   276→
   277→        toast.success('Expense parsed! Review the details and submit.');
   278→      } else {
   279→        toast.error('Could not parse expense. Try being more specific.');
   280→      }
   281→    } catch {
   282→      toast.error('AI is currently unavailable. Please fill the form manually.');
   283→    } finally {
   284→      setNlpLoading(false);
   285→    }
   286→  };
   287→
   288→  // AI categorize
   289→  const handleCategorize = async () => {
   290→    if (!description.trim()) {
   291→      toast.error('Enter a description first');
   292→      return;
   293→    }
   294→    setCategorizeLoading(true);
   295→    try {
   296→      const res = await fetch('/api/ai/categorize', {
   297→        method: 'POST',
   298→        headers: { 'Content-Type': 'application/json' },
   299→        body: JSON.stringify({ description }),
   300→      });
   301→      if (res.ok) {
   302→        const data = await res.json();
   303→        if (data.category) {
   304→          setCategory(data.category);
   305→          toast.success(`Categorized as ${data.category}`);
   306→        }
   307→      }
   308→    } catch {
   309→      toast.error('Failed to categorize');
   310→    } finally {
   311→      setCategorizeLoading(false);
   312→    }
   313→  };
   314→
   315→  // Receipt upload
   316→  const handleReceiptUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   317→    const file = e.target.files?.[0];
   318→    if (!file) return;
   319→    if (file.size > 10 * 1024 * 1024) {
   320→      toast.error('Receipt image must be under 10MB');
   321→      return;
   322→    }
   323→    setReceiptPreview(URL.createObjectURL(file));
   324→    setReceiptLoading(true);
   325→    try {
   326→      const formData = new FormData();
   327→      formData.append('receipt', file);
   328→      const res = await fetch('/api/ai/receipt', { method: 'POST', body: formData });
   329→      if (res.ok) {
   330→        const data = await res.json();
   331→        if (data.description) { setDescription(data.description); setDescError(''); }
   332→        if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   333→        if (data.category) setCategory(data.category);
   334→        if (data.date) setDate(data.date);
   335→        toast.success('Receipt scanned successfully!');
   336→      } else {
   337→        toast.error('Could not read receipt. Try a clearer photo.');
   338→      }
   339→    } catch {
   340→      toast.error('Failed to scan receipt');
   341→    } finally {
   342→      setReceiptLoading(false);
   343→    }
   344→  };
   345→
   346→  // Add email participant
   347→  const addEmailParticipant = () => {
   348→    if (!newEmail.trim()) return;
   349→    const email = newEmail.trim().toLowerCase();
   350→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
   351→    if (!emailRegex.test(email)) {
   352→      toast.error('Please enter a valid email address');
   353→      return;
   354→    }
   355→    if (emailParticipants.some((p) => p.email === email)) {
   356→      toast.error('This email is already added');
   357→      return;
   358→    }
   359→    setEmailParticipants([...emailParticipants, { email, name: newName.trim() || email.split('@')[0] }]);
   360→    setNewEmail('');
   361→    setNewName('');
   362→  };
   363→
   364→  // Remove email participant
   365→  const removeEmail = (email: string) => {
   366→    setEmailParticipants(emailParticipants.filter((p) => p.email !== email));
   367→    setEmailSplitAmounts((prev) => { const n = { ...prev }; delete n[email]; return n; });
   368→  };
   369→
   370→  // Toggle friend selection
   371→  const toggleFriend = (friendId: string) => {
   372→    setSelectedFriends((prev) =>
   373→      prev.includes(friendId) ? prev.filter((id) => id !== friendId) : [...prev, friendId]
   374→    );
   375→  };
   376→
   377→  // Total participants in direct mode
   378→  const totalDirectParticipants = 1 + selectedFriends.length + emailParticipants.length;
   379→  const perPersonDirect = amount && !isNaN(parseFloat(amount)) && totalDirectParticipants > 0
   380→    ? (parseFloat(amount) / totalDirectParticipants).toFixed(2)
   381→    : '0.00';
   382→
   383→  // Update split value
   384→  const updateSplit = (userId: string, value: string) => {
   385→    setSplits((prev) => prev.map((s) => (s.userId === userId ? { ...s, value } : s)));
   386→  };
   387→
   388→  const updateSplitShare = (userId: string, share: number) => {
   389→    setSplits((prev) => prev.map((s) => (s.userId === userId ? { ...s, share: Math.max(1, share) } : s)));
   390→  };
   391→
   392→  // Validate form
   393→  const validate = () => {
   394→    let valid = true;
   395→    if (!description.trim()) {
   396→      setDescError('Please enter what the expense was for');
   397→      valid = false;
   398→    } else {
   399→      setDescError('');
   400→    }
   401→    const numAmount = parseFloat(amount);
   402→    if (!amount || isNaN(numAmount) || numAmount <= 0) {
   403→      setAmountError('Please enter a valid amount greater than ₹0');
   404→      valid = false;
   405→    } else {
   406→      setAmountError('');
   407→    }
   408→    if (mode === 'group' && !groupId) {
   409→      toast.error('Please select a group');
   410→      valid = false;
   411→    }
   412→    if (mode === 'direct' && selectedFriends.length === 0 && emailParticipants.length === 0) {
   413→      toast.error('Select at least one friend or add an email to split with');
   414→      valid = false;
   415→    }
   416→    return valid;
   417→  };
   418→
   419→  // Submit expense
   420→  const handleSubmit = async (e: React.FormEvent) => {
   421→    e.preventDefault();
   422→    if (!validate()) return;
   423→    setSubmitting(true);
   424→    try {
   425→      const body: any = {
   426→        description: description.trim(),
   427→        amount: parseFloat(amount),
   428→        category,
   429→        date,
   430→        paidById,
   431→        splitType,
   432→        note: note.trim(),
   433→      };
   434→
   435→      if (mode === 'group') {
   436→        body.groupId = groupId;
   437→        if (splitType === 'share') {
   438→          // Send share values — backend handles proportional calculation
   439→          body.splitType = 'equal';
   440→          body.splits = splits.map((s) => ({
   441→            userId: s.userId,
   442→            share: s.share || 1,
   443→          }));
   444→        } else if (splitType !== 'equal') {
   445→          body.splits = splits.map((s) => ({
   446→            userId: s.userId,
   447→            value: parseFloat(s.value) || 0,
   448→          }));
   449→        }
   450→      } else {
   451→        // Direct expense
   452→        if (splitType === 'equal') {
   453→          body.splits = selectedFriends.map((id) => ({ userId: id, share: 1 }));
   454→          body.nonUserSplits = emailParticipants.map((p) => ({ email: p.email, name: p.name, share: 1 }));
   455→        } else if (splitType === 'share') {
   456→          body.splitType = 'equal';
   457→          body.splits = splits.map((s) => ({
   458→            userId: s.userId,
   459→            share: s.share || 1,
   460→          }));
   461→          body.nonUserSplits = emailParticipants.map((p) => ({ email: p.email, name: p.name, share: 1 }));
   462→        } else if (splitType === 'exact') {
   463→          body.splits = splits.map((s) => ({
   464→            userId: s.userId,
   465→            amount: parseFloat(s.value) || 0,
   466→          }));
   467→          // Include email participant amounts
   468→          if (emailParticipants.length > 0) {
   469→            body.nonUserSplits = emailParticipants.map((p) => ({
   470→              email: p.email,
   471→              name: p.name,
   472→              amount: parseFloat(emailSplitAmounts[p.email]) || 0,
   473→            }));
   474→          }
   475→        } else if (splitType === 'percentage') {
   476→          body.splits = splits.map((s) => ({
   477→            userId: s.userId,
   478→            percentage: parseFloat(s.value) || 0,
   479→          }));
   480→          // Include email participant percentages
   481→          if (emailParticipants.length > 0) {
   482→            body.nonUserSplits = emailParticipants.map((p) => ({
   483→              email: p.email,
   484→              name: p.name,
   485→              percentage: parseFloat(emailSplitAmounts[p.email]) || 0,
   486→            }));
   487→          }
   488→        }
   489→      }
   490→
   491→      const res = await fetch('/api/expenses', {
   492→        method: 'POST',
   493→        headers: { 'Content-Type': 'application/json' },
   494→        body: JSON.stringify(body),
   495→      });
   496→      if (res.ok) {
   497→        toast.success('Expense added!');
   498→        if (mode === 'group' && groupId) {
   499→          navigateToGroup(groupId);
   500→        } else {
   501→          setView('history');
   502→        }
   503→      } else {
   504→        const data = await res.json();
   505→        toast.error(data.error || 'Failed to add expense. Please try again.');
   506→      }
   507→    } catch {
   508→      toast.error('Network error. Please check your connection.');
   509→    } finally {
   510→      setSubmitting(false);
   511→    }
   512→  };
   513→
   514→  const perPerson = amount && !isNaN(parseFloat(amount)) && members.length > 0
   515→    ? (parseFloat(amount) / members.length).toFixed(2)
   516→    : '0.00';
   517→
   518→  // Split participants for direct exact/percentage mode
   519→  const directParticipants = [
   520→    ...(friends.filter((f) => selectedFriends.includes(f.id)).map((f) => ({ id: f.id, name: f.name }))),
   521→  ];
   522→
   523→  useEffect(() => {
   524→    if (mode === 'direct' && (splitType === 'exact' || splitType === 'percentage' || splitType === 'share')) {
   525→      setSplits(directParticipants.map((p) => ({ userId: p.id, value: '', share: 1 })));
   526→    }
   527→  }, [mode, splitType, selectedFriends.length]);
   528→
   529→  return (
   530→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} className="max-w-2xl mx-auto space-y-5">
   531→      {/* Header */}
   532→      <div>
   533→        <h2 className="text-xl font-bold text-gray-900">Add Expense</h2>
   534→        <p className="text-sm text-gray-500 mt-1">Record a new expense and split it.</p>
   535→      </div>
   536→
   537→      {/* AI Natural Language Input */}
   538→      <Card className="border-emerald-200 bg-gradient-to-r from-emerald-50/50 to-teal-50/30">
   539→        <CardContent className="p-4">
   540→          <div className="flex items-center gap-2 mb-2.5">
   541→            <Sparkles className="w-4 h-4 text-emerald-600" />
   542→            <h3 className="font-semibold text-gray-900 text-sm">Quick Add with AI</h3>
   543→            <TooltipProvider>
   544→              <Tooltip>
   545→                <TooltipTrigger>
   546→                  <Info className="w-3.5 h-3.5 text-gray-400" />
   547→                </TooltipTrigger>
   548→                <TooltipContent className="text-xs max-w-xs">
   549→                  Describe your expense in plain English. AI will extract the amount, category, and split details.
   550→                  <br /><br />
   551→                  Examples: &quot;Paid ₹450 for pizza split equally with Alex and Sam&quot;
   552→                </TooltipContent>
   553→              </Tooltip>
   554→            </TooltipProvider>
   555→          </div>
   556→          <div className="flex gap-2">
   557→            <div className="flex-1">
   558→              <Input
