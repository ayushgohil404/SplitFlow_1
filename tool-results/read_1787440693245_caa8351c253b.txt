    77→export function AddExpenseView() {
    78→  const { selectedGroupId, navigateToGroup, setView, user } = useAppStore();
    79→
    80→  // Expense mode
    81→  const [mode, setMode] = useState<ExpenseMode>(selectedGroupId ? 'group' : 'direct');
    82→
    83→  // Data
    84→  const [groups, setGroups] = useState<Group[]>([]);
    85→  const [members, setMembers] = useState<Member[]>([]);
    86→  const [friends, setFriends] = useState<Friend[]>([]);
    87→  const [membersLoading, setMembersLoading] = useState(false);
    88→
    89→  // AI NLP input
    90→  const [nlpInput, setNlpInput] = useState('');
    91→  const [nlpLoading, setNlpLoading] = useState(false);
    92→  // Stores parsed friend split amounts (name → value) from AI parse, applied after useEffect
    93→  const [pendingFriendSplitValues, setPendingFriendSplitValues] = useState<Record<string, string>>({});
    94→
    95→  // Form fields
    96→  const [description, setDescription] = useState('');
    97→  const [amount, setAmount] = useState('');
    98→  const [amountError, setAmountError] = useState('');
    99→  const [descError, setDescError] = useState('');
   100→  const [category, setCategory] = useState('general');
   101→  const [date, setDate] = useState(new Date().toISOString().split('T')[0]);
   102→  const [groupId, setGroupId] = useState(selectedGroupId || '');
   103→  const [paidById, setPaidById] = useState(user?.id || '');
   104→  const [splitType, setSplitType] = useState<'equal' | 'share' | 'exact' | 'percentage'>('equal');
   105→  const [splits, setSplits] = useState<{ userId: string; value: string; share: number }[]>([]);
   106→  const [note, setNote] = useState('');
   107→
   108→  // Direct expense: selected friends + email participants
   109→  const [selectedFriends, setSelectedFriends] = useState<string[]>([]);
   110→  const [emailParticipants, setEmailParticipants] = useState<EmailParticipant[]>([]);
   111→  const [emailSplitAmounts, setEmailSplitAmounts] = useState<Record<string, string>>({});
   112→  const [userSplitValue, setUserSplitValue] = useState('');
   113→  const [newEmail, setNewEmail] = useState('');
   114→  const [newName, setNewName] = useState('');
   115→
   116→  // Receipt upload
   117→  const [receiptPreview, setReceiptPreview] = useState<string | null>(null);
   118→  const [receiptLoading, setReceiptLoading] = useState(false);
   119→
   120→  // AI categorize
   121→  const [categorizeLoading, setCategorizeLoading] = useState(false);
   122→
   123→  // Submitting
   124→  const [submitting, setSubmitting] = useState(false);
   125→
   126→  // Fetch groups
   127→  useEffect(() => {
   128→    async function fetchData() {
   129→      try {
   130→        const [groupsRes, friendsRes] = await Promise.all([
   131→          fetch('/api/groups'),
   132→          fetch('/api/friends'),
   133→        ]);
   134→        if (groupsRes.ok) {
   135→          const data = await groupsRes.json();
   136→          const list = Array.isArray(data) ? data : data.groups || [];
   137→          setGroups(list);
   138→          if (!selectedGroupId && list.length > 0) {
   139→            setGroupId(list[0].id);
   140→          }
   141→        }
   142→        if (friendsRes.ok) {
   143→          const fData = await friendsRes.json();
   144→          setFriends(fData.friends || []);
   145→        }
   146→      } catch {
   147→        // silent
   148→      }
   149→    }
   150→    fetchData();
   151→  }, [selectedGroupId]);
   152→
   153→  // Fetch group members when group changes
   154→  useEffect(() => {
   155→    if (mode !== 'group' || !groupId) return;
   156→    setMembersLoading(true);
   157→    async function fetchMembers() {
   158→      try {
   159→        const res = await fetch(`/api/groups/${groupId}`);
   160→        if (res.ok) {
   161→          const data = await res.json();
   162→          const g = data.group || data;
   163→          const memberList = ((g.members || []) as any[]).map((m: any) => ({
   164→            id: m.user?.id || m.userId || m.id,
   165→            name: m.user?.name || m.name || 'Unknown',
   166→            email: m.user?.email || m.email || '',
   167→          }));
   168→          setMembers(memberList);
   169→          setSplits(memberList.map((m: Member) => ({ userId: m.id, value: '', share: 1 })));
   170→          const me = memberList.find((m: Member) => m.id === user?.id);
   171→          if (me) setPaidById(me.id);
   172→        }
   173→      } catch {
   174→        // silent
   175→      } finally {
   176→        setMembersLoading(false);
   177→      }
   178→    }
   179→    fetchMembers();
   180→  }, [groupId, mode, user?.id]);
   181→
   182→  // AI parse expense
   183→  const handleNlpSubmit = async () => {
   184→    if (!nlpInput.trim()) return;
   185→    setNlpLoading(true);
   186→    // Clear pending values from previous parse
   187→    setPendingFriendSplitValues({});
   188→    try {
   189→      const res = await fetch('/api/ai/parse-expense', {
   190→        method: 'POST',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify({ text: nlpInput }),
   193→      });
   194→      const data = await res.json();
   195→
   196→      // Handle HTTP errors and AI errors uniformly
   197→      if (!res.ok && res.status !== 200) {
   198→        toast.error(data.error || 'Could not parse expense. Try being more specific.');
   199→        return;
   200→      }
   201→      if (data.error) {
   202→        console.log('[AI Parse] Error response:', data);
   203→        if (data.code === 'NOT_CONFIGURED') {
   204→          toast.error('AI is not configured. Please set a valid GROQ_API_KEY in Vercel environment variables.');
   205→        } else if (data.code === 'AI_ERROR') {
   206→          toast.error(data.error);
   207→        } else {
   208→          toast.error(data.error || 'Could not parse expense. Try a different phrasing.');
   209→        }
   210→        return;
   211→      }
   212→
   213→      // Set basic fields
   214→      if (data.description) { setDescription(data.description); setDescError(''); }
   215→      if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   216→      if (data.category) setCategory(data.category);
   217→      if (data.date) setDate(data.date);
   218→
   219→      const sType = data.splitType as string;
   220→      const isExactOrPct = sType === 'exact' || sType === 'percentage';
   221→
   222→      // Handle group selection from AI parse
   223→      if (data.groupId && typeof data.groupId === 'string') {
   224→        const groupName = data.groupId.toLowerCase();
   225→        const matchedGroup = groups.find((g) => g.name.toLowerCase() === groupName);
   226→        if (matchedGroup) {
   227→          setMode('group');
   228→          setGroupId(matchedGroup.id);
   229→          // Don't process friend/email splits for group expenses
   230→          if (isExactOrPct) {
   231→            setSplitType(sType as 'exact' | 'percentage');
   232→          } else if (sType === 'equal' || sType === 'share') {
   233→            setSplitType(sType === 'share' ? 'share' : 'equal');
   234→          } else if (sType === 'single') {
   235→            setSplitType('equal');
   236→          }
   237→          toast.success(`Expense parsed for group "${matchedGroup.name}"! Review and submit.`);
   238→          return;
   239→        }
   240→        // Group name not found — fall through to direct mode
   241→      }
   242→
   243→      // Determine mode: switch to direct if there are email splits or named non-me splits
   244→      const emailSplits = data.emailSplits as { email: string; name?: string; amount?: number; percentage?: number }[] | undefined;
   245→      const namedSplits = data.splits as { name: string; amount?: number | null; percentage?: number | null; share?: number }[] | undefined;
   246→      const hasEmails = emailSplits && emailSplits.length > 0;
   247→      const hasNamedSplits = namedSplits && namedSplits.some((s) => s.name !== 'me');
   248→
   249→      // Handle "single" / "only me" / personal expenses — no participants
   250→      if (sType === 'single' && !hasEmails && !hasNamedSplits) {
   251→        setMode('direct');
   252→        setSelectedFriends([]);
   253→        setEmailParticipants([]);
   254→        setEmailSplitAmounts({});
   255→        setPendingFriendSplitValues({});
   256→        setSplitType('equal');
   257→        toast.success('Expense parsed! No split needed — just submit.');
   258→        return;
   259→      }
   260→
   261→      // Extract "me" split value for exact/percentage modes
   262→      if (isExactOrPct) {
   263→        const meSplit = namedSplits?.find((s) => s.name === 'me');
   264→        if (meSplit && (meSplit.amount != null || meSplit.percentage != null)) {
   265→          setUserSplitValue(String(meSplit.amount ?? meSplit.percentage));
   266→        } else if (data.amount) {
   267→          const totalAmt = parseFloat(String(data.amount));
   268→          let othersTotal = 0;
   269→          if (sType === 'exact') {
   270→            for (const ns of (namedSplits || []).filter((s) => s.name !== 'me')) othersTotal += ns.amount ?? 0;
   271→            for (const es of (emailSplits || [])) othersTotal += es.amount ?? 0;
   272→          } else {
   273→            for (const ns of (namedSplits || []).filter((s) => s.name !== 'me')) othersTotal += ns.percentage ?? 0;
   274→            for (const es of (emailSplits || [])) othersTotal += es.percentage ?? 0;
   275→          }
   276→          const myValue = sType === 'exact'
   277→            ? Math.round((totalAmt - othersTotal) * 100) / 100
   278→            : Math.round((100 - othersTotal) * 100) / 100;
   279→          setUserSplitValue(myValue > 0 ? String(myValue) : '');
   280→        } else {
   281→          setUserSplitValue('');
   282→        }
   283→      } else {
   284→        setUserSplitValue('');
   285→      }
   286→
   287→      // Switch to direct mode when we have participants to add
   288→      if (hasEmails || hasNamedSplits) {
   289→        setMode('direct');
   290→      }
   291→
   292→      // Process email splits
   293→      if (hasEmails) {
   294→        const newEmailParts: EmailParticipant[] = emailSplits.map((es) => ({
   295→          email: es.email,
   296→          name: es.name || es.email.split('@')[0],
   297→        }));
   298→        setEmailParticipants(newEmailParts);
   299→
   300→        if (isExactOrPct) {
   301→          const amtMap: Record<string, string> = {};
   302→          for (const es of emailSplits) {
   303→            if (es.amount != null) amtMap[es.email] = String(es.amount);
   304→            else if (es.percentage != null) amtMap[es.email] = String(es.percentage);
   305→          }
   306→          setEmailSplitAmounts(amtMap);
   307→        } else {
   308→          setEmailSplitAmounts({});
   309→        }
   310→      } else {
   311→        setEmailParticipants([]);
   312→        setEmailSplitAmounts({});
   313→      }
   314→
   315→      // Process named splits — match to friends or add as email participants
   316→      const friendAmtMap: Record<string, string> = {};
   317→      const nonMeSplits = (namedSplits || []).filter((s) => s.name !== 'me');
   318→
   319→      for (const ns of nonMeSplits) {
   320→        const friendMatch = friends.find(
   321→          (f) =>
   322→            f.name?.toLowerCase().trim() === ns.name?.toLowerCase().trim() ||
   323→            f.email?.toLowerCase().split('@')[0] === ns.name?.toLowerCase().trim() ||
   324→            f.email?.toLowerCase() === ns.name?.toLowerCase()
   325→        );
   326→        if (friendMatch) {
   327→          if (!selectedFriends.includes(friendMatch.id)) {
   328→            setSelectedFriends((prev) => [...prev, friendMatch.id]);
   329→          }
   330→          // For share type, set the friend's share value
   331→          if (sType === 'share' && ns.share != null && ns.share > 0) {
   332→            // We'll apply share values after the useEffect sets up splits
   333→            friendAmtMap[friendMatch.id] = String(ns.share);
   334→          } else if (isExactOrPct && (ns.amount != null || ns.percentage != null)) {
   335→            friendAmtMap[friendMatch.id] = String(ns.amount ?? ns.percentage);
   336→          }
   337→        } else if (ns.name?.includes('@')) {
   338→          // Named person looks like an email — add as email participant
   339→          const emailAddr = ns.name.toLowerCase();
   340→          if (!emailSplits?.some((es) => es.email.toLowerCase() === emailAddr)) {
   341→            setEmailParticipants((prev) => [...prev, { email: emailAddr, name: ns.name.split('@')[0] }]);
   342→            if (isExactOrPct && (ns.amount != null || ns.percentage != null)) {
   343→              setEmailSplitAmounts((prev) => ({ ...prev, [emailAddr]: String(ns.amount ?? ns.percentage) }));
   344→            }
   345→          }
   346→        }
   347→        // If not a friend and not an email, we skip them — user can add manually
   348→        // (non-registered users need an email to track)
   349→      }
   350→
   351→      if (Object.keys(friendAmtMap).length > 0) {
   352→        setPendingFriendSplitValues(friendAmtMap);
   353→      } else {
   354→        setPendingFriendSplitValues({});
   355→      }
   356→
   357→      // Handle split type
   358→      if (isExactOrPct) {
   359→        setSplitType(sType as 'exact' | 'percentage');
   360→      } else if (sType === 'share') {
   361→        setSplitType('share');
   362→        // Set owner share from AI parse if available
   363→        const meSplit = namedSplits?.find((s) => s.name === 'me');
   364→        if (meSplit?.share && typeof meSplit.share === 'number' && meSplit.share > 0) {
   365→          setOwnerShare(meSplit.share);
   366→        }
   367→      } else if (sType === 'equal') {
   368→        setSplitType('equal');
   369→      } else if (sType === 'single') {
   370→        setSplitType('equal');
   371→      }
   372→
   373→      toast.success('Expense parsed! Review the details and submit.');
   374→    } catch (err: any) {
   375→      console.error('[AI Parse] Client error:', err);
   376→      toast.error('Network error. Please check your connection and try again.');
   377→    } finally {
   378→      setNlpLoading(false);
   379→    }
   380→  };
   381→
   382→  // AI categorize
   383→  const handleCategorize = async () => {
   384→    if (!description.trim()) {
   385→      toast.error('Enter a description first');
   386→      return;
   387→    }
   388→    setCategorizeLoading(true);
   389→    try {
   390→      const res = await fetch('/api/ai/categorize', {
   391→        method: 'POST',
   392→        headers: { 'Content-Type': 'application/json' },
   393→        body: JSON.stringify({ description }),
   394→      });
   395→      if (res.ok) {
   396→        const data = await res.json();
   397→        if (data.error) {
   398→          toast.error(data.error || 'Failed to categorize');
   399→        } else if (data.category) {
   400→          setCategory(data.category);
   401→          toast.success(`Categorized as ${data.category}`);
   402→        }
   403→      } else {
   404→        const data = await res.json();
   405→        toast.error(data.error || 'Failed to categorize');
   406→      }
   407→    } catch {
   408→      toast.error('Failed to categorize');
   409→    } finally {
   410→      setCategorizeLoading(false);
   411→    }
   412→  };
   413→
   414→  // Receipt upload
   415→  const handleReceiptUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   416→    const file = e.target.files?.[0];
   417→    if (!file) return;
   418→    if (file.size > 10 * 1024 * 1024) {
   419→      toast.error('Receipt image must be under 10MB');
   420→      return;
   421→    }
   422→    setReceiptPreview(URL.createObjectURL(file));
   423→    setReceiptLoading(true);
   424→    try {
   425→      const formData = new FormData();
   426→      formData.append('receipt', file);
   427→      const res = await fetch('/api/ai/receipt', { method: 'POST', body: formData });
   428→      if (res.ok) {
   429→        const data = await res.json();
   430→        if (data.description) { setDescription(data.description); setDescError(''); }
   431→        if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   432→        if (data.category) setCategory(data.category);
   433→        if (data.date) setDate(data.date);
   434→        toast.success('Receipt scanned successfully!');
   435→      } else {
   436→        toast.error('Could not read receipt. Try a clearer photo.');
   437→      }
   438→    } catch {
   439→      toast.error('Failed to scan receipt');
   440→    } finally {
   441→      setReceiptLoading(false);
   442→    }
   443→  };
   444→
   445→  // Add email participant
   446→  const addEmailParticipant = () => {
   447→    if (!newEmail.trim()) return;
   448→    const email = newEmail.trim().toLowerCase();
   449→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
   450→    if (!emailRegex.test(email)) {
   451→      toast.error('Please enter a valid email address');
   452→      return;
   453→    }
   454→    if (emailParticipants.some((p) => p.email === email)) {
   455→      toast.error('This email is already added');
   456→      return;
   457→    }
   458→    setEmailParticipants([...emailParticipants, { email, name: newName.trim() || email.split('@')[0] }]);
   459→    setNewEmail('');
   460→    setNewName('');
   461→  };
   462→
   463→  // Remove email participant
   464→  const removeEmail = (email: string) => {
   465→    setEmailParticipants(emailParticipants.filter((p) => p.email !== email));
   466→    setEmailSplitAmounts((prev) => { const n = { ...prev }; delete n[email]; return n; });
   467→  };
   468→
   469→  // Toggle friend selection
   470→  const toggleFriend = (friendId: string) => {
   471→    setSelectedFriends((prev) =>
   472→      prev.includes(friendId) ? prev.filter((id) => id !== friendId) : [...prev, friendId]
   473→    );
   474→  };
   475→
   476→  // Total participants in direct mode
   477→  const totalDirectParticipants = 1 + selectedFriends.length + emailParticipants.length;
   478→  const perPersonDirect = amount && !isNaN(parseFloat(amount)) && totalDirectParticipants > 0
   479→    ? (parseFloat(amount) / totalDirectParticipants).toFixed(2)
   480→    : '0.00';
   481→
   482→  // Update split value
   483→  const updateSplit = (userId: string, value: string) => {
   484→    setSplits((prev) => prev.map((s) => (s.userId === userId ? { ...s, value } : s)));
   485→  };
   486→
   487→  const updateSplitShare = (userId: string, share: number) => {
   488→    setSplits((prev) => prev.map((s) => (s.userId === userId ? { ...s, share: Math.max(1, share) } : s)));
   489→  };
   490→
   491→  // Validate form
   492→  const validate = () => {
   493→    let valid = true;
   494→    if (!description.trim()) {
   495→      setDescError('Please enter what the expense was for');
   496→      valid = false;
   497→    } else {
   498→      setDescError('');
   499→    }
   500→    const numAmount = parseFloat(amount);
   501→    if (!amount || isNaN(numAmount) || numAmount <= 0) {
   502→      setAmountError('Please enter a valid amount greater than ₹0');
   503→      valid = false;
   504→    } else {
   505→      setAmountError('');
   506→    }
   507→    if (mode === 'group' && !groupId) {
   508→      toast.error('Please select a group');
   509→      valid = false;
   510→    }
   511→    if (mode === 'direct' && selectedFriends.length === 0 && emailParticipants.length === 0 && splitType !== 'equal') {
   512→      toast.error('Select at least one friend or add an email to split with');
   513→      valid = false;
   514→    }
   515→    // For equal split with no participants, auto-switch to owner-only (no splits needed)
   516→    if (mode === 'direct' && selectedFriends.length === 0 && emailParticipants.length === 0 && splitType === 'equal') {
   517→      // Personal expense — submit without splits
   518→    }
   519→    return valid;
   520→  };
   521→
   522→  // Submit expense
   523→  const handleSubmit = async (e: React.FormEvent) => {
   524→    e.preventDefault();
   525→    if (!validate()) return;
   526→    setSubmitting(true);
   527→    try {
   528→      const body: any = {
   529→        description: description.trim(),
   530→        amount: parseFloat(amount),
   531→        category,
   532→        date,
   533→        paidById,
   534→        splitType,
   535→        note: note.trim(),
   536→      };
   537→
   538→      if (mode === 'group') {
   539→        body.groupId = groupId;
   540→        if (splitType === 'share') {
   541→          // Send share values — backend handles proportional calculation
   542→          body.splitType = 'equal';
   543→          body.splits = splits.map((s) => ({
   544→            userId: s.userId,
   545→            share: s.share || 1,
   546→          }));
   547→        } else if (splitType === 'exact') {
   548→          body.splits = splits.map((s) => ({
   549→            userId: s.userId,
   550→            amount: parseFloat(s.value) || 0,
