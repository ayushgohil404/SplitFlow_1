    75→    75→
    76→    76→export function AddExpenseView() {
    77→    77→  const { selectedGroupId, navigateToGroup, setView, user } = useAppStore();
    78→    78→
    79→    79→  // Expense mode
    80→    80→  const [mode, setMode] = useState<ExpenseMode>(selectedGroupId ? 'group' : 'direct');
    81→    81→
    82→    82→  // Data
    83→    83→  const [groups, setGroups] = useState<Group[]>([]);
    84→    84→  const [members, setMembers] = useState<Member[]>([]);
    85→    85→  const [friends, setFriends] = useState<Friend[]>([]);
    86→    86→  const [membersLoading, setMembersLoading] = useState(false);
    87→    87→
    88→    88→  // AI NLP input
    89→    89→  const [nlpInput, setNlpInput] = useState('');
    90→    90→  const [nlpLoading, setNlpLoading] = useState(false);
    91→    91→
    92→    92→  // Form fields
    93→    93→  const [description, setDescription] = useState('');
    94→    94→  const [amount, setAmount] = useState('');
    95→    95→  const [amountError, setAmountError] = useState('');
    96→    96→  const [descError, setDescError] = useState('');
    97→    97→  const [category, setCategory] = useState('general');
    98→    98→  const [date, setDate] = useState(new Date().toISOString().split('T')[0]);
    99→    99→  const [groupId, setGroupId] = useState(selectedGroupId || '');
   100→   100→  const [paidById, setPaidById] = useState(user?.id || '');
   101→   101→  const [splitType, setSplitType] = useState<'equal' | 'exact' | 'percentage'>('equal');
   102→   102→  const [splits, setSplits] = useState<{ userId: string; value: string }[]>([]);
   103→   103→  const [note, setNote] = useState('');
   104→   104→
   105→   105→  // Direct expense: selected friends + email participants
   106→   106→  const [selectedFriends, setSelectedFriends] = useState<string[]>([]);
   107→   107→  const [emailParticipants, setEmailParticipants] = useState<EmailParticipant[]>([]);
   108→   108→  const [newEmail, setNewEmail] = useState('');
   109→   109→  const [newName, setNewName] = useState('');
   110→   110→
   111→   111→  // Receipt upload
   112→   112→  const [receiptPreview, setReceiptPreview] = useState<string | null>(null);
   113→   113→  const [receiptLoading, setReceiptLoading] = useState(false);
   114→   114→
   115→   115→  // AI categorize
   116→   116→  const [categorizeLoading, setCategorizeLoading] = useState(false);
   117→   117→
   118→   118→  // Submitting
   119→   119→  const [submitting, setSubmitting] = useState(false);
   120→   120→
   121→   121→  // Fetch groups
   122→   122→  useEffect(() => {
   123→   123→    async function fetchData() {
   124→   124→      try {
   125→   125→        const [groupsRes, friendsRes] = await Promise.all([
   126→   126→          fetch('/api/groups'),
   127→   127→          fetch('/api/friends'),
   128→   128→        ]);
   129→   129→        if (groupsRes.ok) {
   130→   130→          const data = await groupsRes.json();
   131→   131→          const list = Array.isArray(data) ? data : data.groups || [];
   132→   132→          setGroups(list);
   133→   133→          if (!selectedGroupId && list.length > 0) {
   134→   134→            setGroupId(list[0].id);
   135→   135→          }
   136→   136→        }
   137→   137→        if (friendsRes.ok) {
   138→   138→          const fData = await friendsRes.json();
   139→   139→          setFriends(fData.friends || []);
   140→   140→        }
   141→   141→      } catch {
   142→   142→        // silent
   143→   143→      }
   144→   144→    }
   145→   145→    fetchData();
   146→   146→  }, [selectedGroupId]);
   147→   147→
   148→   148→  // Fetch group members when group changes
   149→   149→  useEffect(() => {
   150→   150→    if (mode !== 'group' || !groupId) return;
   151→   151→    setMembersLoading(true);
   152→   152→    async function fetchMembers() {
   153→   153→      try {
   154→   154→        const res = await fetch(`/api/groups/${groupId}`);
   155→   155→        if (res.ok) {
   156→   156→          const data = await res.json();
   157→   157→          const g = data.group || data;
   158→   158→          const memberList = ((g.members || []) as any[]).map((m: any) => ({
   159→   159→            id: m.user?.id || m.userId || m.id,
   160→   160→            name: m.user?.name || m.name || 'Unknown',
   161→   161→            email: m.user?.email || m.email || '',
   162→   162→          }));
   163→   163→          setMembers(memberList);
   164→   164→          setSplits(memberList.map((m: Member) => ({ userId: m.id, value: '' })));
   165→   165→          const me = memberList.find((m: Member) => m.id === user?.id);
   166→   166→          if (me) setPaidById(me.id);
   167→   167→        }
   168→   168→      } catch {
   169→   169→        // silent
   170→   170→      } finally {
   171→   171→        setMembersLoading(false);
   172→   172→      }
   173→   173→    }
   174→   174→    fetchMembers();
   175→   175→  }, [groupId, mode, user?.id]);
   176→   176→
   177→   177→  // AI parse expense
   178→   178→  const handleNlpSubmit = async () => {
   179→   179→    if (!nlpInput.trim()) return;
   180→   180→    setNlpLoading(true);
   181→   181→    try {
   182→   182→      const res = await fetch('/api/ai/parse-expense', {
   183→   183→        method: 'POST',
   184→   184→        headers: { 'Content-Type': 'application/json' },
   185→   185→        body: JSON.stringify({ text: nlpInput }),
   186→   186→      });
   187→   187→      if (res.ok) {
   188→   188→        const data = await res.json();
   189→   189→        if (data.description) { setDescription(data.description); setDescError(''); }
   190→   190→        if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   191→   191→        if (data.category) setCategory(data.category);
   192→   192→        if (data.date) setDate(data.date);
   193→   193→        if (data.splitType) setSplitType(data.splitType);
   194→   194→        toast.success('Expense parsed from your description!');
   195→   195→      } else {
   196→   196→        toast.error('Could not parse expense. Try being more specific.');
   197→   197→      }
   198→   198→    } catch {
   199→   199→      toast.error('AI is currently unavailable. Please fill the form manually.');
   200→   200→    } finally {
   201→   201→      setNlpLoading(false);
   202→   202→    }
   203→   203→  };
   204→   204→
   205→   205→  // AI categorize
   206→   206→  const handleCategorize = async () => {
   207→   207→    if (!description.trim()) {
   208→   208→      toast.error('Enter a description first');
   209→   209→      return;
   210→   210→    }
   211→   211→    setCategorizeLoading(true);
   212→   212→    try {
   213→   213→      const res = await fetch('/api/ai/categorize', {
   214→   214→        method: 'POST',
   215→   215→        headers: { 'Content-Type': 'application/json' },
   216→   216→        body: JSON.stringify({ description }),
   217→   217→      });
   218→   218→      if (res.ok) {
   219→   219→        const data = await res.json();
   220→   220→        if (data.category) {
   221→   221→          setCategory(data.category);
   222→   222→          toast.success(`Categorized as ${data.category}`);
   223→   223→        }
   224→   224→      }
   225→   225→    } catch {
   226→   226→      toast.error('Failed to categorize');
   227→   227→    } finally {
   228→   228→      setCategorizeLoading(false);
   229→   229→    }
   230→   230→  };
   231→   231→
   232→   232→  // Receipt upload
   233→   233→  const handleReceiptUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   234→   234→    const file = e.target.files?.[0];
   235→   235→    if (!file) return;
   236→   236→    if (file.size > 10 * 1024 * 1024) {
   237→   237→      toast.error('Receipt image must be under 10MB');
   238→   238→      return;
   239→   239→    }
   240→   240→    setReceiptPreview(URL.createObjectURL(file));
   241→   241→    setReceiptLoading(true);
   242→   242→    try {
   243→   243→      const formData = new FormData();
   244→   244→      formData.append('receipt', file);
   245→   245→      const res = await fetch('/api/ai/receipt', { method: 'POST', body: formData });
   246→   246→      if (res.ok) {
   247→   247→        const data = await res.json();
   248→   248→        if (data.description) { setDescription(data.description); setDescError(''); }
   249→   249→        if (data.amount) { setAmount(String(data.amount)); setAmountError(''); }
   250→   250→        if (data.category) setCategory(data.category);
   251→   251→        if (data.date) setDate(data.date);
   252→   252→        toast.success('Receipt scanned successfully!');
   253→   253→      } else {
   254→   254→        toast.error('Could not read receipt. Try a clearer photo.');
   255→   255→      }
   256→   256→    } catch {
   257→   257→      toast.error('Failed to scan receipt');
   258→   258→    } finally {
   259→   259→      setReceiptLoading(false);
   260→   260→    }
   261→   261→  };
   262→   262→
   263→   263→  // Add email participant
   264→   264→  const addEmailParticipant = () => {
   265→   265→    if (!newEmail.trim()) return;
   266→   266→    const email = newEmail.trim().toLowerCase();
   267→   267→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
   268→   268→    if (!emailRegex.test(email)) {
   269→   269→      toast.error('Please enter a valid email address');
   270→   270→      return;
   271→   271→    }
   272→   272→    if (emailParticipants.some((p) => p.email === email)) {
   273→   273→      toast.error('This email is already added');
   274→   274→      return;
   275→   275→    }
   276→   276→    setEmailParticipants([...emailParticipants, { email, name: newName.trim() || email.split('@')[0] }]);
   277→   277→    setNewEmail('');
   278→   278→    setNewName('');
   279→   279→  };
   280→   280→
   281→   281→  // Remove email participant
   282→   282→  const removeEmail = (email: string) => {
   283→   283→    setEmailParticipants(emailParticipants.filter((p) => p.email !== email));
   284→   284→  };
   285→   285→
   286→   286→  // Toggle friend selection
   287→   287→  const toggleFriend = (friendId: string) => {
   288→   288→    setSelectedFriends((prev) =>
   289→   289→      prev.includes(friendId) ? prev.filter((id) => id !== friendId) : [...prev, friendId]
   290→   290→    );
   291→   291→  };
   292→   292→
   293→   293→  // Total participants in direct mode
   294→   294→  const totalDirectParticipants = 1 + selectedFriends.length + emailParticipants.length;
   295→   295→  const perPersonDirect = amount && !isNaN(parseFloat(amount)) && totalDirectParticipants > 0
   296→   296→    ? (parseFloat(amount) / totalDirectParticipants).toFixed(2)
   297→   297→    : '0.00';
   298→   298→
   299→   299→  // Update split value
   300→   300→  const updateSplit = (userId: string, value: string) => {
   301→   301→    setSplits((prev) => prev.map((s) => (s.userId === userId ? { ...s, value } : s)));
   302→   302→  };
   303→   303→
   304→   304→  // Validate form
   305→   305→  const validate = () => {
   306→   306→    let valid = true;
   307→   307→    if (!description.trim()) {
   308→   308→      setDescError('Please enter what the expense was for');
   309→   309→      valid = false;
   310→   310→    } else {
   311→   311→      setDescError('');
   312→   312→    }
   313→   313→    const numAmount = parseFloat(amount);
   314→   314→    if (!amount || isNaN(numAmount) || numAmount <= 0) {
   315→   315→      setAmountError('Please enter a valid amount greater than ₹0');
   316→   316→      valid = false;
   317→   317→    } else {
   318→   318→      setAmountError('');
   319→   319→    }
   320→   320→    if (mode === 'group' && !groupId) {
   321→   321→      toast.error('Please select a group');
   322→   322→      valid = false;
   323→   323→    }
   324→   324→    if (mode === 'direct' && selectedFriends.length === 0 && emailParticipants.length === 0) {
   325→   325→      toast.error('Select at least one friend or add an email to split with');
   326→   326→      valid = false;
   327→   327→    }
   328→   328→    return valid;
   329→   329→  };
   330→   330→
   331→   331→  // Submit expense
   332→   332→  const handleSubmit = async (e: React.FormEvent) => {
   333→   333→    e.preventDefault();
   334→   334→    if (!validate()) return;
   335→   335→    setSubmitting(true);
   336→   336→    try {
   337→   337→      const body: any = {
   338→   338→        description: description.trim(),
   339→   339→        amount: parseFloat(amount),
   340→   340→        category,
   341→   341→        date,
   342→   342→        paidById,
   343→   343→        splitType,
   344→   344→        note: note.trim(),
   345→   345→      };
   346→   346→
   347→   347→      if (mode === 'group') {
   348→   348→        body.groupId = groupId;
   349→   349→        if (splitType !== 'equal') {
   350→   350→          body.splits = splits.map((s) => ({
   351→   351→            userId: s.userId,
   352→   352→            value: parseFloat(s.value) || 0,
   353→   353→          }));
   354→   354→        }
   355→   355→      } else {
   356→   356→        // Direct expense
   357→   357→        if (splitType === 'equal') {
   358→   358→          body.splits = selectedFriends.map((id) => ({ userId: id, share: 1 }));
   359→   359→          body.nonUserSplits = emailParticipants.map((p) => ({ email: p.email, name: p.name, share: 1 }));
   360→   360→        } else if (splitType === 'exact') {
   361→   361→          body.splits = splits.map((s) => ({
   362→   362→            userId: s.userId,
   363→   363→            amount: parseFloat(s.value) || 0,
   364→   364→          }));
   365→   365→        } else {
   366→   366→          body.splits = splits.map((s) => ({
   367→   367→            userId: s.userId,
   368→   368→            percentage: parseFloat(s.value) || 0,
   369→   369→          }));
   370→   370→        }
   371→   371→      }
   372→   372→
   373→   373→      const res = await fetch('/api/expenses', {
   374→   374→        method: 'POST',
   375→   375→        headers: { 'Content-Type': 'application/json' },
   376→   376→        body: JSON.stringify(body),
   377→   377→      });
   378→   378→      if (res.ok) {
   379→   379→        toast.success('Expense added!');
   380→   380→        if (mode === 'group' && groupId) {
   381→   381→          navigateToGroup(groupId);
   382→   382→        } else {
   383→   383→          setView('history');
   384→   384→        }
   385→   385→      } else {
   386→   386→        const data = await res.json();
   387→   387→        toast.error(data.error || 'Failed to add expense. Please try again.');
   388→   388→      }
   389→   389→    } catch {
   390→   390→      toast.error('Network error. Please check your connection.');
   391→   391→    } finally {
   392→   392→      setSubmitting(false);
   393→   393→    }
   394→   394→  };
   395→   395→
   396→   396→  const perPerson = amount && !isNaN(parseFloat(amount)) && members.length > 0
   397→   397→    ? (parseFloat(amount) / members.length).toFixed(2)
   398→   398→    : '0.00';
   399→   399→
   400→   400→  // Split participants for direct exact/percentage mode
   401→   401→  const directParticipants = [
   402→   402→    ...(friends.filter((f) => selectedFriends.includes(f.id)).map((f) => ({ id: f.id, name: f.name }))),
   403→   403→  ];
   404→   404→
   405→   405→  useEffect(() => {
   406→   406→    if (mode === 'direct' && splitType !== 'equal') {
   407→   407→      setSplits(directParticipants.map((p) => ({ userId: p.id, value: '' })));
   408→   408→    }
   409→   409→  }, [mode, splitType, selectedFriends.length]);
   410→   410→
   411→   411→  return (
   412→   412→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} className="max-w-2xl mx-auto space-y-5">
   413→   413→      {/* Header */}
   414→   414→      <div>
   415→   415→        <h2 className="text-xl font-bold text-gray-900">Add Expense</h2>
   416→   416→        <p className="text-sm text-gray-500 mt-1">Record a new expense and split it.</p>
   417→   417→      </div>
   418→   418→
   419→   419→      {/* AI Natural Language Input */}
   420→   420→      <Card className="border-emerald-200 bg-gradient-to-r from-emerald-50/50 to-teal-50/30">
   421→   421→        <CardContent className="p-4">
   422→   422→          <div className="flex items-center gap-2 mb-2.5">
   423→   423→            <Sparkles className="w-4 h-4 text-emerald-600" />
   424→   424→            <h3 className="font-semibold text-gray-900 text-sm">Quick Add with AI</h3>
   425→   425→            <TooltipProvider>
   426→   426→              <Tooltip>
   427→   427→                <TooltipTrigger>
   428→   428→                  <Info className="w-3.5 h-3.5 text-gray-400" />
   429→   429→                </TooltipTrigger>
   430→   430→                <TooltipContent className="text-xs max-w-xs">
   431→   431→                  Describe your expense in plain English. AI will extract the amount, category, and split details.
   432→   432→                  <br /><br />
   433→   433→                  Examples: &quot;Paid ₹450 for pizza split equally with Alex and Sam&quot;
   434→   434→                </TooltipContent>
   435→   435→              </Tooltip>
   436→   436→            </TooltipProvider>
   437→   437→          </div>
   438→   438→          <div className="flex gap-2">
   439→   439→            <div className="flex-1">
   440→   440→              <Input
   441→   441→                placeholder='e.g. "Paid ₹450 for pizza split 3 ways"'
   442→   442→                value={nlpInput}
   443→   443→                onChange={(e) => setNlpInput(e.target.value)}
   444→   444→                onKeyDown={(e) => e.key === 'Enter' && handleNlpSubmit()}
   445→   445→                className="h-11 bg-white"
   446→   446→                disabled={nlpLoading}
   447→   447→              />
   448→   448→            </div>
   449→   449→            <Button
   450→   450→              onClick={handleNlpSubmit}
   451→   451→              disabled={nlpLoading || !nlpInput.trim()}
   452→   452→              className="bg-emerald-600 hover:bg-emerald-700 text-white h-11 px-4 shrink-0"
   453→   453→            >
   454→   454→              {nlpLoading ? <Loader2 className="w-4 h-4 animate-spin" /> : <Sparkles className="w-4 h-4" />}
   455→   455→            </Button>
   456→   456→          </div>
   457→   457→        </CardContent>
   458→   458→      </Card>
   459→   459→
   460→   460→      {/* Main Form */}
   461→   461→      <Card>
   462→   462→        <CardContent className="p-5 sm:p-6">
   463→   463→          <form onSubmit={handleSubmit} className="space-y-5" noValidate>
   464→   464→            {/* Expense Mode Toggle */}
   465→   465→            <div className="space-y-2">
   466→   466→              <Label className="text-sm font-medium">Expense Type</Label>
   467→   467→              <div className="grid grid-cols-2 gap-3">
   468→   468→                <button
   469→   469→                  type="button"
   470→   470→                  onClick={() => setMode('direct')}
   471→   471→                  className={`flex items-center justify-center gap-2 px-4 py-3 rounded-lg border-2 transition-all text-sm font-medium ${
   472→   472→                    mode === 'direct'
   473→   473→                      ? 'border-emerald-500 bg-emerald-50 text-emerald-700'
